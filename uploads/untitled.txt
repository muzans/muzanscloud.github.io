<div class="container">
    <h3>Masukkan Link</h3>
    <div class="input-wrapper">
        <input type="text" id="linkInput" placeholder="Tempel link di sini...">
        <button class="btn-paste" onclick="pasteLink()">Paste</button>
    </div>
    
    <button id="btn-connect" onclick="connectAction()">Connect</button>
</div>

<script>
    const inputField = document.getElementById('linkInput');
    const connectBtn = document.getElementById('btn-connect');

    // Fungsi untuk memantau perubahan pada kolom input
    inputField.addEventListener('input', function() {
        if (inputField.value.trim() !== "") {
            connectBtn.classList.add('visible');
        } else {
            connectBtn.classList.remove('visible');
        }
    });

    // Fungsi untuk fitur Paste
    async function pasteLink() {
        try {
            const text = await navigator.clipboard.readText();
            inputField.value = text;
            
            // Trigger event input secara manual agar tombol Connect muncul
            inputField.dispatchEvent(new Event('input'));
        } catch (err) {
            alert('Gagal menyalin teks. Pastikan Anda memberi izin akses clipboard.');
        }
    }

    function connectAction() {
        alert("Menghubungkan ke: " + inputField.value);
    }
</script>
